    def _find_matching_instances(self, client, in_spaces=None, existence_match="equals"):
        # with existence_match="normalized", look for the same ROR ID before looking for the same name
        if existence_match == "normalized":
            instances = self._find_by_identifiers(client, ("RORID",), in_spaces=in_spaces)
            if instances:
